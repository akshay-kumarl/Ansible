---
- hosts: all
  become: yes		{this line is for giving root permissions}   
  tasks:
   - name: install Apache 
     apt: name=apache2 state=latest
   - name: copy index.html
     copy: src=index.html dest=/var/www/html
   - name: restart apache server 
     service: name=apache2 state=restarted
---

Dry run 
ansible-playbook apache.yml --check

To run playbook 
ansible-playbook apache.yml
ansible-playbook [name of the playbook]




notify,handlers 
---
- hosts: all	
  become: yes
  tasks:
   - name: install apache2
     apt: name=apache2 state=latest 
   - name: copy index file
     copy: src=index.html dest=/var/www/html
     notify: Restart apache2
 
  handlers:
   - name: Restart apache2
     service: name=apache2 state=restarted  
---


  

register: to capture the output of the tasks
debug: to display the output 


---
- hosts: all
  become: yes
  tasks:
    - name: install apache
      apt: name=apache2 state=latest
      register: apache_install_result # register result of the task
    
    - name: Debug apache installation Result 
      debug:
        var: apache_install_result   # Print the result captured in the register variable

    - name: copy index.html
      copy: src=index.html dest=/var/www/html
      notify: restart apache2

  handlers:
    - name: Restart apache2
      service: name=apache2 state=restarted
      
---

    name: Techie

=================================
Varsfile example
cat variables.yml 
---
name: devops
---

Playbook file

---
- name: variable playbook
  hosts: all
  vars_files:
     - variables.yml   {pass the path of the variables file}   

  tasks:
    - name: Display value
      debug:
        var: name
---

ansible-playbook variable.yml --syntax-check

    - name: create directory

        path: /opt/{{item}}

       - directory1

       - directory3
       - directory4
